read "examples/data/grouped-sales.csv" into sales
summarize sales by region with total = sum(revenue) skipping missing, mean = average(revenue) skipping missing, rows = count(revenue) into totals
print totals
filter totals where total > 25 into top_regions
print top_regions

# Group on combinations, then roll the result back up to region totals.
summarize sales by region, product with total = sum(revenue) skipping missing, rows = count(revenue) into detail
print detail
summarize detail by region with total = sum(total) into rolled_up
print rolled_up
